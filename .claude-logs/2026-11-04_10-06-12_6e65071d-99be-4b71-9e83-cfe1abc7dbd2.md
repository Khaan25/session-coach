// Throwaway, not part of the repo. Terminal output is counts only, never any log text.
// Usage (from the repo root):
//   node --env-file=apps/web/.env.local --experimental-strip-types --disable-warning=ExperimentalWarning ~/eval-cards.mts <label> <log1.jsonl> <log2.jsonl> ...
// <label> is "old" or "new": just a name for the output file. The code comes from the repo in the current directory.
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { register } from "node:module";
import { pathToFileURL } from "node:url";

const repo = process.cwd();
const web = `${repo}/apps/web`;
const pipeline = `${web}/src/lib/pipeline`;

// call-model.ts imports "server-only" and "@/..." aliases that plain node can't resolve. This resolver maps "@/" to
// apps/web/src and turns "server-only" into an empty module, so the real callModel loads unchanged.
const hooks = `
export async function initialize(data) { globalThis.__src = data.src; }
export async function resolve(specifier, context, next) {
  if (specifier === "server-only") return { url: "data:text/javascript,", shortCircuit: true };
  if (specifier.startsWith("@/")) {
    const base = globalThis.__src + "/" + specifier.slice(2);
    for (const candidate of [base + ".ts", base + "/index.ts", base]) {
      try { return await next(new URL("file://" + candidate).href, context); } catch {}
    }
  }
  return next(specifier, context);
}`;
register(`data:text/javascript,${encodeURIComponent(hooks)}`, {
  parentURL: pathToFileURL(`${repo}/`).href,
  data: { src: `${web}/src` },
});

// callModel retries a 429 inside itself, so a 429 that it recovered from would be invisible. Count them at fetch.
let count429 = 0;
const realFetch = globalThis.fetch;
globalThis.fetch = async (...args: Parameters<typeof fetch>) => {
  const response = await realFetch(...args);
  if (response.status === 429) count429++;
  return response;
};

const { parseSession } = await import(`${pipeline}/parser.ts`);
const { trimTimeline } = await import(`${pipeline}/process-rows.ts`);
const { extractCards } = await import(`${pipeline}/extract.ts`);
const { checkEvidence } = await import(`${pipeline}/evidence.ts`);
const { callModel } = await import(`${web}/src/lib/model/call-model.ts`);

const [label, ...logs] = process.argv.slice(2);
if (!label || logs.length === 0) {
  console.error("usage: eval-cards.mts <label> <log1.jsonl> <log2.jsonl> ...");
  process.exit(1);
}

const WAIT_MS = 65_000;
const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

type Ev = { id: string; kind: string; text: string; line: number };
type Card = {
  decision: string;
  why: string;
  question: string;
  model_answer: string;
};
type Result = {
  path: string;
  events: number;
  keptEvents: number;
  firstPromptKept: boolean | null;
  returned: number | null;
  keptCards: number | null;
  failed: boolean;
  errorName: string | null;
  had429: boolean;
  cards: Card[];
};

const FIELDS = ["decision", "why", "question", "model_answer"] as const;
const BACKTICK = /`/;
const IDENTIFIER = [
  /\b[a-z][a-z0-9]*[A-Z][A-Za-z0-9]*\b/, // camelCase
  /\b[a-z0-9]+_[a-z0-9_]+\b/i, // snake_case
  /(?:^|[\s("'])~?\.{0,2}\/?[\w.@-]+\/[\w./@-]+/, // a path
  /\b[\w-]+\.(?:tsx?|jsx?|mjs|mts|cjs|json|md|py|sql|css|html|ya?ml|toml|sh|lock|env)\b/i, // a file name
];
const ASKS_WHY = /\bwhy\b/i;
const ASKS_WHAT_CHECK = /\bwhat\s+does\b[^?]*\bcheck/i;

const words = (text: string) => text.trim().split(/\s+/).filter(Boolean).length;
const fieldsMatching = (card: Card, test: (text: string) => boolean) =>
  FIELDS.filter((field) => test(card[field]));
const yesNo = (value: boolean | null) => (value === null ? "n/a" : value ? "yes" : "no");

async function evalLog(path: string): Promise<Result> {
  const result: Result = {
    path,
    events: 0,
    keptEvents: 0,
    firstPromptKept: null,
    returned: null,
    keptCards: null,
    failed: false,
    errorName: null,
    had429: false,
    cards: [],
  };
  const before429 = count429;
  try {
    const { events } = parseSession(readFileSync(path, "utf8")) as { events: Ev[] };
    result.events = events.length;
    // Same order as processSession: parse, trim, extract (default level), check evidence.
    const { kept: timeline } = trimTimeline(events) as { kept: Ev[] };
    result.keptEvents = timeline.length;
    const firstPrompt = events.find((event) => event.kind === "prompt");
    result.firstPromptKept = firstPrompt ? timeline.some((event) => event.id === firstPrompt.id) : null;

    const drafts = await extractCards(timeline, callModel);
    result.returned = drafts.length;
    const cards = checkEvidence(drafts, timeline) as Card[];
    result.keptCards = cards.length;
    result.cards = cards;
  } catch (error) {
    result.failed = true;
    result.errorName = error instanceof Error ? error.name : typeof error;
  }
  result.had429 = count429 > before429;
  return result;
}

function countsLine(i: number, r: Result) {
  return [
    `log ${i + 1}:`,
    `events=${r.events}`,
    `kept=${r.keptEvents}`,
    `first_prompt_kept=${yesNo(r.firstPromptKept)}`,
    `cards_returned=${r.returned ?? "n/a"}`,
    `cards_kept=${r.keptCards ?? "n/a"}`,
    `failed=${r.failed ? `yes (${r.errorName})` : "no"}`,
    `429=${yesNo(r.had429)}`,
  ].join(" ");
}

const quote = (text: string) =>
  text
    .split("\n")
    .map((line) => `> ${line}`)
    .join("\n");

function section(i: number, r: Result) {
  const lines = [
    `## Log ${i + 1}: ${r.path}`,
    "",
    `- events: ${r.events}`,
    `- events kept after trim: ${r.keptEvents}`,
    `- first prompt kept: ${yesNo(r.firstPromptKept)}`,
    `- cards the model returned: ${r.returned ?? "n/a"}`,
    `- cards kept after checkEvidence: ${r.keptCards ?? "n/a"}`,
    `- call failed: ${r.failed ? `yes (${r.errorName})` : "no"}`,
    `- 429 happened: ${yesNo(r.had429)}`,
    "",
  ];
  r.cards.forEach((card, n) => {
    const identifierFields = fieldsMatching(card, (text) => IDENTIFIER.some((re) => re.test(text)));
    const backtickFields = fieldsMatching(card, (text) => BACKTICK.test(text));
    const list = (fields: readonly string[]) => (fields.length ? `yes (${fields.join(", ")})` : "no");
    lines.push(
      `### Card ${n + 1}`,
      "",
      "**decision**",
      quote(card.decision),
      "",
      "**why**",
      quote(card.why),
      "",
      "**question**",
      quote(card.question),
      "",
      "**model_answer**",
      quote(card.model_answer),
      "",
      "Automatic flags:",
      "",
      `- contains backticks: ${list(backtickFields)}`,
      `- contains a code identifier or file name: ${list(identifierFields)}`,
      `- word count: decision ${words(card.decision)}, why ${words(card.why)}, question ${words(card.question)}, model_answer ${words(card.model_answer)}`,
      `- question asks "why": ${yesNo(ASKS_WHY.test(card.question))}`,
      `- question asks "what does ... check": ${yesNo(ASKS_WHAT_CHECK.test(card.question))}`,
      "",
      "Your check:",
      "",
      "- [ ] a real decision",
      "- [ ] readable on a phone",
      "- [ ] worth asking",
      "",
    );
  });
  return lines.join("\n");
}

const date = new Date();
const stamp = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}-${String(date.getDate()).padStart(2, "0")}`;
const outDir = `${process.env.HOME}/eval-out`;
const outPath = `${outDir}/${stamp}-${label}.md`;
mkdirSync(outDir, { recursive: true });

const results: Result[] = [];
const write = () =>
  writeFileSync(
    outPath,
    [`# eval-cards: ${label} (${stamp})`, "", ...results.map((r, i) => section(i, r))].join("\n"),
  );

for (const [i, path] of logs.entries()) {
  if (i > 0) await sleep(WAIT_MS);
  const result = await evalLog(path);
  results.push(result);
  console.log(countsLine(i, result));
  write(); // after every log, so a crash keeps what is done
}
console.log("wrote", outPath);
